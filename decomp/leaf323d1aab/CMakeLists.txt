cmake_minimum_required(VERSION 3.16)
project(re5_leaf323d1aab LANGUAGES CXX)
include(CTest)
file(GLOB LEAF_SOURCES CONFIGURE_DEPENDS "generated/src/leaves_*.cpp")
if(NOT LEAF_SOURCES)
  message(FATAL_ERROR "Missing generated recovery source")
endif()
add_library(re5_leaf323d1aab STATIC ${LEAF_SOURCES} generated/src/catalog.cpp)
target_compile_features(re5_leaf323d1aab PUBLIC cxx_std_17)
target_include_directories(re5_leaf323d1aab PUBLIC generated/include)
if(MSVC)
  target_compile_options(re5_leaf323d1aab PRIVATE /W4 /WX)
else()
  target_compile_options(re5_leaf323d1aab PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(re5_leaf_properties tests/properties.cpp)
target_link_libraries(re5_leaf_properties PRIVATE re5_leaf323d1aab)
add_test(NAME leaf_source_properties COMMAND re5_leaf_properties)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME leaf_catalogue_integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_package.py)
add_test(NAME leaf_reject_wrong_input COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_input.py)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  add_executable(re5_leaf_native tests/native_fragments.cpp)
  target_link_libraries(re5_leaf_native PRIVATE re5_leaf323d1aab)
  # Deliberately NOT registered in ordinary CI: the copyrighted EXE stays local.
endif()
add_test(NAME leaf_membership_parser_synthetic COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/membership_fixture.py)
add_test(NAME leaf_reproducible_sources COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/restore_sources.py --check)
if(UNIX)
  add_executable(re5_leaf_source_guards tests/source_guards.cpp)
  target_link_libraries(re5_leaf_source_guards PRIVATE re5_leaf323d1aab)
  add_test(NAME leaf_source_guard_pages COMMAND re5_leaf_source_guards)
endif()
