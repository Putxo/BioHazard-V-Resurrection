cmake_minimum_required(VERSION 3.20)
project(RE5Sequences323d1aab LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(re5_sequence323d1aab STATIC src/sequences.cpp)
target_include_directories(re5_sequence323d1aab PUBLIC include)
target_compile_features(re5_sequence323d1aab PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(re5_sequence323d1aab PRIVATE /W4 /WX)
else()
  target_compile_options(re5_sequence323d1aab PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME sequence_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify.py --self-test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  add_executable(re5_sequence_tests tests/test_sequences.cpp)
  target_link_libraries(re5_sequence_tests PRIVATE re5_sequence323d1aab)
  target_compile_options(re5_sequence_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME sequence_properties COMMAND re5_sequence_tests properties)
  add_test(NAME sequence_guards COMMAND re5_sequence_tests guards)
endif()
