cmake_minimum_required(VERSION 3.20)
project(RE5Scalar323d1aab LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
file(GLOB scalar_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated/src/scalars_*.cpp")
add_library(re5_scalar323d1aab STATIC ${scalar_sources} generated/src/catalog.cpp)
target_include_directories(re5_scalar323d1aab PUBLIC generated/include)
target_compile_features(re5_scalar323d1aab PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(re5_scalar323d1aab PRIVATE /W4 /WX)
else()
  target_compile_options(re5_scalar323d1aab PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME scalar_reproducible_sources COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate.py --check)
add_test(NAME scalar_reject_wrong_input COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_input.py)
# Fixed-address mmap/guard fixtures are Linux host tests, not game execution.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  foreach(test properties guards)
    add_executable(re5_scalar_${test} tests/${test}.cpp)
    target_link_libraries(re5_scalar_${test} PRIVATE re5_scalar323d1aab)
    target_compile_options(re5_scalar_${test} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME scalar_${test} COMMAND re5_scalar_${test})
  endforeach()
endif()
