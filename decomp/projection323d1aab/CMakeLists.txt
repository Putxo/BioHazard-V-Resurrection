cmake_minimum_required(VERSION 3.20)
project(RE5Projection323d1aab LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(re5_projection323d1aab STATIC src/projections.cpp)
target_include_directories(re5_projection323d1aab PUBLIC include)
target_compile_features(re5_projection323d1aab PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(re5_projection323d1aab PRIVATE /W4 /WX)
else()
  target_compile_options(re5_projection323d1aab PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME projection_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify.py --self-test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  add_executable(re5_projection_tests tests/test_projections.cpp)
  target_link_libraries(re5_projection_tests PRIVATE re5_projection323d1aab)
  target_compile_options(re5_projection_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME projection_properties COMMAND re5_projection_tests properties)
  add_test(NAME projection_guards COMMAND re5_projection_tests guards)
endif()
