cmake_minimum_required(VERSION 3.20)
project(RE5Transfers323d1aab LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS records.csv tools/model.py tools/verify.py)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify.py --emit ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
add_library(re5_transfer323d1aab STATIC src/transfers.cpp)
target_include_directories(re5_transfer323d1aab PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(re5_transfer323d1aab PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(re5_transfer323d1aab PRIVATE /W4 /WX)
else()
  target_compile_options(re5_transfer323d1aab PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME transfer_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify.py --self-test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  add_executable(re5_transfer_tests tests/test_transfers.cpp)
  target_link_libraries(re5_transfer_tests PRIVATE re5_transfer323d1aab)
  target_compile_options(re5_transfer_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME transfer_properties COMMAND re5_transfer_tests properties)
  add_test(NAME transfer_guards COMMAND re5_transfer_tests guards)
endif()
